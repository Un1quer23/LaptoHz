cmake_minimum_required(VERSION 3.25)
project(LaptoHz VERSION 0.4.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This application requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  add_link_options(-static -static-libstdc++ -static-libgcc)
endif()
add_library(rrs_core src/policy.cpp)
target_include_directories(rrs_core PUBLIC src)
add_library(rrs_windows src/display.cpp src/physical_modes.cpp src/support.cpp src/localization.cpp src/startup.cpp src/tray_menu.cpp src/mode_hint.cpp src/appearance.cpp)
target_link_libraries(rrs_windows PUBLIC rrs_core user32 gdi32 shell32 advapi32 ole32 powrprof wtsapi32 dwmapi uuid runtimeobject comctl32 uxtheme)
add_executable(LaptoHz WIN32 src/main.cpp src/app.cpp src/popup.cpp resources/app.rc)
set_property(SOURCE resources/app.rc APPEND PROPERTY OBJECT_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest" "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico"
  "${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-light.ico" "${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-dark.ico")
target_link_libraries(LaptoHz PRIVATE rrs_windows gdiplus)
if(MSVC)
  target_link_options(LaptoHz PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(LaptoHz PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(localization_tests tests/localization_tests.cpp)
  target_link_libraries(localization_tests PRIVATE rrs_windows)
  add_test(NAME localization COMMAND localization_tests)
  add_executable(policy_tests tests/policy_tests.cpp)
  target_link_libraries(policy_tests PRIVATE rrs_core)
  add_test(NAME policy_tests COMMAND policy_tests)
  add_executable(confirmation_tests tests/confirmation_tests.cpp)
  target_link_libraries(confirmation_tests PRIVATE rrs_core)
  add_test(NAME confirmation_tests COMMAND confirmation_tests)
  add_executable(physical_modes_tests tests/physical_modes_tests.cpp)
  target_link_libraries(physical_modes_tests PRIVATE rrs_windows)
  add_test(NAME physical_modes_tests COMMAND physical_modes_tests)
  add_executable(display_recovery_tests tests/display_recovery_tests.cpp)
  target_link_libraries(display_recovery_tests PRIVATE rrs_windows)
  add_test(NAME display_recovery COMMAND display_recovery_tests)
  add_executable(settings_tests tests/settings_tests.cpp)
  target_link_libraries(settings_tests PRIVATE rrs_windows)
  add_test(NAME settings_tests COMMAND settings_tests)
  add_executable(log_path_tests tests/log_path_tests.cpp)
  target_link_libraries(log_path_tests PRIVATE rrs_windows)
  add_test(NAME diagnostic_log_path COMMAND log_path_tests)
  add_executable(startup_tests tests/startup_tests.cpp)
  target_link_libraries(startup_tests PRIVATE rrs_windows)
  add_test(NAME startup_registration COMMAND startup_tests)
  add_executable(popup_tests tests/popup_tests.cpp src/popup.cpp)
  target_link_libraries(popup_tests PRIVATE rrs_windows gdiplus)

  set_tests_properties(mode_hint_ui PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)

  target_link_libraries(controller_tests PRIVATE rrs_windows gdiplus)
  add_test(NAME controller_integration COMMAND controller_tests)
  set_tests_properties(controller_integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 90)
  add_executable(display_smoke_test tests/display_smoke_test.cpp)
  target_link_libraries(display_smoke_test PRIVATE rrs_windows)
  add_test(NAME display_read_only COMMAND display_smoke_test)
  set_tests_properties(display_read_only PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(runtime_smoke_test tests/runtime_smoke_test.cpp)
  target_link_libraries(runtime_smoke_test PRIVATE rrs_windows)

install(TARGETS LaptoHz RUNTIME DESTINATION .)
install(FILES README.md README.en.md LICENSE DESTINATION .)
